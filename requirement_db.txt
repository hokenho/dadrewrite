# Tables Administration
# ---------------------------------- Workday tables ----------------------------------
# User Information Table (sync from workday)
CREATE TABLE wd_user_profile (
    gad_id        NVARCHAR(255) PRIMARY KEY,
    employee_id   NUMBER,
    last_name     NVARCHAR(255),
    first_name    NVARCHAR(255),
    signer_cost_center     NVARCHAR(255),
    job_level_id  NUMBER FK, (-> wd_job_level.id)
    territory_id  NUMBER FK, (-> wd_territory.id)
    status       BIT
    last_update_date  DATE
)
# Job Levels (source=Workday)
CREATE TABLE wd_job_level (
    id           NUMBER PRIMARY KEY,
    job_grade    NVARCHAR(255),
    job_title    NVARCHAR(MAX),    # assume title could have japanese characters
    limit_amount_max	INTEGER,
    status       BIT
    last_update_date  DATE
)
# Countries/Territories (source=Workday)
CREATE TABLE wd_territory (
    id              NUMBER PRIMARY KEY,
    country_code    NVARCHAR(255),
    territory       NVARCHAR(255),
    status       BIT
    last_update_date  DATE
)
# ---------------------------------- GDAS tables ----------------------------------
# Policies Categories (source=GDAS)
CREATE TABLE gdas_policy (
    id                  NUMBER PRIMARY KEY,
    country_code        NVARCHAR(255),
    category            NVARCHAR(255),
    description         NVARCHAR(255),
    system              NVARCHAR(255), # 3 values only (AP/SNOW/Concur) -> store directly
    approval_count      NUMBER(1), #3 values only (1,2,3) -> store directly
    threshold_amount    NUMBER,
    manual_approval     BIT,
    status       BIT
    last_update_date  DATE
)

# ---------------------------------- Infor tables ----------------------------------
# Business Units (source=Infor)
CREATE TABLE infor_business_unit (
    id              NUMBER PRIMARY KEY,
    security_group  NVARCHAR(255),
    name            NVARCHAR(255),
    status       BIT
    last_update_date  DATE
)
# Cost Centres (source=Infor)
CREATE TABLE infor_cost_center (
    id           NUMBER PRIMARY KEY,
    name         NVARCHAR(255),
    description  NVARCHAR(255),
    status       BIT
    last_update_date  DATE
)
# Currencies (source=Infor)
CREATE TABLE infor_currency (
    currency     NVARCHAR(255) PRIMARY KEY,
    name         NVARCHAR(255),
    status       BIT
    last_update_date  DATE
)
# Exchange Rates (source=Infor)
CREATE TABLE infor_exchange_rate (
    source_currency       NVARCHAR(255),
    target_currency       NVARCHAR(255),
    plan_rate             FLOAT,
    status       BIT,
    last_update_date  DATE
    PRIMARY KEY (source, target)  # Composite PK - using both source and target currency
)

# Tables for Signer approval
CREATE TABLE signer (
    gad_id NVARCHAR(255) PRIMARY KEY, (->wd_user_profile.gad_id)
    bu_id NUMBER FK (->infor_business_unit.id)
    pickup_cheque BIT,
    temp BIT,
    temp_start_date DATE,
    temp_end_date DATE,
    status      BIT
)
CREATE TABLE signer_limit (
    id      NUMBER PRIMARY KEY,
    gad_id  NVARCHAR(255),  (-> signer.gad_id 1:M signer_limit.gad_id)
    territory_id  NUMBER FK, (-> wd_territory.id)
    policy_id     NUMBER FK, (-> gdas_policy.id)
    limit_amount  NUMBER,
    approval_cc_id   NUMBER FK, (-> infor_cost_center.id)
    exception     BIT,
    temp          BIT,
    temp_start_date DATE,
    temp_end_date   DATE,
    business_rationale     BIT,
    business_control       BIT,
    account_restriction    NVARCHAR(100),
    currency    NVARCHAR(10)
)

# Tables for Active DAD Request
CREATE TABLE req_signer (
    id          NUMBER PRIMARY KEY,
    req_date    DATE,
    submitter   NVARCHAR(255) FK, (-> wd_user_profile.gad_id),
    gad_id NVARCHAR(255), (->wd_user_profile.gad_id)
    bu_id NUMBER FK (->infor_business_unit.id)
    approver1_gad_id NVARCHAR(255),
    approver1_result BIT NULLABLE, # NULL=pending approval,0=reject,1=approve
    approver1_note NVARCHAR(255), 
    approver2_gad_id NVARCHAR(255),
    approver2_result BIT NULLABLE, # NULL=pending approval,0=reject,1=approve
    approver2_note NVARCHAR(255),

    # old signer values (snapshot at time of request submission)
    old_pickup_cheque BIT,
    old_temp BIT,
    old_temp_start_date DATE,
    old_temp_end_date DATE,
    old_status BIT,

    # new requested signer values
    new_pickup_cheque BIT,
    new_temp BIT,
    new_temp_start_date DATE,
    new_temp_end_date DATE,
    new_status      BIT     # NULL=pending approval,0=reject,1=approve
    status_reason   NVARCHAR(255)
)

CREATE TABLE req_signer_limit (
    id            NUMBER PRIMARY KEY,
    req_signer_id NUMBER, (-> req_signer.id 1:M req_signer_limit.req_signer_id)
    action        NVARCHAR(255), # 4 values only (Change/Add/Remove/Unchange) -> store directly
    signer_limit_id   NUMBER FK, (-> signer_limit.id. For Change/Remove/Unchange only, else NULL)

    # old signer_limit values (snapshot at time of request submission). For Change/Remove/Unchange only, else NULL
    old_territory_id  NUMBER FK, (-> wd_territory.id)
    old_policy_id     NUMBER FK, (-> gdas_policy.id)
    old_limit_amount  NUMBER,
    old_approval_cc_id   NUMBER FK, (-> infor_cost_center.id)
    old_exception     BIT,
    old_temp          BIT,
    old_temp_start_date DATE,
    old_temp_end_date   DATE,
    old_business_rationale     BIT,
    old_business_control       BIT,
    old_account_restriction    NVARCHAR(100),
    old_currency    NVARCHAR(10),

    # new requested signer_limit values. For Change/Add only, else NULL
    new_territory_id  NUMBER FK, (-> wd_territory.id)
    new_policy_id     NUMBER FK, (-> gdas_policy.id)
    new_limit_amount  NUMBER,
    new_approval_cc_id   NUMBER FK, (-> infor_cost_center.id)
    new_exception     BIT,
    new_temp          BIT,
    new_temp_start_date DATE,
    new_temp_end_date   DATE,
    new_business_rationale     BIT,
    new_business_control       BIT,
    new_account_restriction    NVARCHAR(100),
    new_currency    NVARCHAR(10)
)

# Tables for Audit/History
# Audited tables: wd_user_profile, wd_job_level, wd_territory, gdas_policy,
#   infor_business_unit, infor_cost_center, infor_currency, infor_exchange_rate,
#   user_access, signer, signer_limit
# NOT audited: req_signer, req_signer_limit (already snapshot old/new values)
# Rules: INSERT -> old_* = NULL; DELETE -> new_* = NULL; UPDATE -> both populated
# FKs are expanded to underlying display fields (no FK IDs stored)

# Audit: wd_user_profile (FKs expanded: job_level_id, territory_id)

    old_last_update_date   DATE,

    new_last_name          NVARCHAR(255),
    new_first_name         NVARCHAR(255),
    new_signer_cost_center NVARCHAR(255),
    new_job_grade          NVARCHAR(255),    # expanded from job_level_id
    new_job_title          NVARCHAR(MAX),    # expanded from job_level_id
