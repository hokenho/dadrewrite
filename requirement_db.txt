# Tables Administration
# ---------------------------------- Workday tables ----------------------------------
# User Information Table (sync from workday)
CREATE TABLE user_profile (
    gad_id        NVARCHAR(255) PRIMARY KEY,
    employee_id   NUMBER,
    last_name     NVARCHAR(255),
    first_name    NVARCHAR(255),
    signer_cc     NVARCHAR(255),
    job_level_id  NUMBER FK, (-> job_level.id)
    territory_id  NUMBER FK (-> territory.id)
)
# Job Levels (source=Workday)
CREATE TABLE job (
    id           NUMBER PRIMARY KEY,
    job_grade    NVARCHAR(255),
    job_title    NVARCHAR(MAX),    # assume title could have japanese characters
    limit_amount	INTEGER,
)
# Countries/Territories (source=Workday)
CREATE TABLE territory (
    id              NUMBER PRIMARY KEY,
    country_code    NVARCHAR(255),
    territory       NVARCHAR(255)
)
# ---------------------------------- GDAS tables ----------------------------------
# Policies Categories (source=GDAS)
CREATE TABLE policy (
    id                  NUMBER PRIMARY KEY,
    country             NVARCHAR(255),
    category            NVARCHAR(255),
    description         NVARCHAR(255),
    system              NVARCHAR(255), # 3 values only (AP/SNOW/Concur) -> store directly
    approval_no         NUMBER(1), #3 values only (1,2,3) -> store directly
    threshold_amount    NUMBER,
    manual_approval     BIT
)

# ---------------------------------- Infor tables ----------------------------------
# Business Units (source=Infor)
CREATE TABLE business_unit (
    id              NUMBER PRIMARY KEY,
    security_group  NVARCHAR(255),
    name            NVARCHAR(255)
)
# Cost Centres (source=Infor)
CREATE TABLE cost_center (
    id           NUMBER PRIMARY KEY,
    cc           NVARCHAR(255),
    description  NVARCHAR(255),
    status       BIT
)
# Currencies (source=Infor)
CREATE TABLE currency (
    currency     NVARCHAR(255) PRIMARY KEY,
    name         NVARCHAR(255)
)
# Exchange Rates (source=Infor)
CREATE TABLE exchange_rate (
    source_currency       NVARCHAR(255),
    target_currency       NVARCHAR(255),
    rate                  FLOAT,
    PRIMARY KEY (source, target)  # Composite PK - using both source and target currency
)

# Tables for Signer approval
CREATE TABLE signer (
    gad_id NVARCHAR(255) PRIMARY KEY, (->user_profile.gad_id)
    bu_id NUMBER FK (->business_unit.id)
    pickup_cheque BIT,
    temp BIT,
    temp_start_date DATE,
    temp_end_date DATE,
    status      BIT
)
CREATE TABLE signer_limit (
    id      NUMBER PRIMARY KEY,
    gad_id  NVARCHAR(255),  (-> signer.gad_id 1:M signer_limit.gad_id)
    territory_id  NUMBER FK, (-> territory.id)
    policy_id     NUMBER FK, (-> policy.id)
    limit_amount  NUMBER,
    approval_cc_id   NUMBER FK, (-> cost_center.id)
    exception     BIT,
    temp          BIT,
    temp_start_date DATE,
    temp_end_date   DATE,
    business_rationale     BIT,
    business_control       BIT,
    account_restriction    NVARCHAR(100),
    currency    NVARCHAR(10)
)

# Tables for Active DAD Request
CREATE TABLE signer_req (
    id          NUMBER PRIMARY KEY,
    req_date    DATE,
    submitter   NVARCHAR(255) FK, (-> user_profile.gad_id),
    gad_id NVARCHAR(255), (->user_profile.gad_id)
    bu_id NUMBER FK (->business_unit.id)
    approver_avp_gad_id NVARCHAR(255),
    approver_avp_result BIT NULLABLE, # NULL=pending approval,0=reject,1=approve
    approver_avp_reason NVARCHAR(255), 
    approver_tsa_gad_id NVARCHAR(255),
    approver_tsa_result BIT NULLABLE, # NULL=pending approval,0=reject,1=approve
    approver_tsa_reason NVARCHAR(255),

    # storing pickup_cheque, temp, temp_start_date, temp_end_date, status signer data in json format when submit the request
    source_data     NVARCHAR(MAX),       

    # below fields format is the same as signer, but only save the new values
    pickup_cheque BIT,
    temp BIT,
    temp_start_date DATE,
    temp_end_date DATE,
    status      BIT     # NULL=pending approval,0=reject,1=approve
    status_reason   NVARCHAR(255)
)

CREATE TABLE signer_limit_req (
    id            NUMBER PRIMARY KEY,
    signer_req_id NUMBER, (-> signer_req.id 1:M signer_limit_req.signer_req_id)
    action        NVARCHAR(255), # 3 values only (Change/Add/Remove) -> store directly
    source_id     NUMBER FK, (-> signer_limit.id, NULL for action=Add)

    # storing the source_id row data in json format when submit the request
    source_data   NVARCHAR(MAX)  

    # below fields format is the same as signer_limit, but only save the new values for the source_id row. For Add/Remove action they should be NULL
    territory_id  NUMBER FK, (-> territory.id)
    policy_id     NUMBER FK, (-> policy.id)
    limit_amount  NUMBER,
    approval_cc_id   NUMBER FK, (-> cost_center.id)
    exception     BIT,
    temp          BIT,
    temp_start_date DATE,
    temp_end_date   DATE,
    business_rationale     BIT,
    business_control       BIT,
    account_restriction    NVARCHAR(100),
    currency    NVARCHAR(10)
)

# Table for Audit/History - dadreq
CREATE TABLE audit_dadreq (
    id          NUMBER PRIMARY KEY,
    gad_id      NVARCHAR(255),
    req_date    DATE,
    submitter   NVARCHAR(255),
    status      NVARCHAR(255), # 3 values only (Complete/Reject/Cancel) -> store directly
    reason      NVARCHAR(MAX), 
    profile_old NVARCHAR(MAX), # save the old data in json format
    profile_new NVARCHAR(MAX), # save the new data in json format
    limit_old   NVARCHAR(MAX), # save the old data in json format
    limit_new   NVARCHAR(MAX)  # save the new data in json format
    audit_date  DATE
)
CREATE INDEX idx_audit_dadreq_gad_id ON audit_dadreq(gad_id);
CREATE INDEX idx_audit_dadreq_status ON audit_dadreq(status);
CREATE INDEX idx_audit_dadreq_date ON audit_dadreq(audit_date);

# Table for Audit/History - all other tables
CREATE TABLE audit_change (
    id          NUMBER PRIMARY KEY,
    audit_table NVARCHAR(50),
    data_old    NVARCHAR(MAX) # save the old data in json format
    data_new    NVARCHAR(MAX) # save the new data in json format
    audit_date  DATE
)
CREATE INDEX idx_audit_change_table ON audit_change(audit_table);
CREATE INDEX idx_audit_change_date ON audit_change(audit_date);
